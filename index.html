function submitClaim() { const week = document.getElementById('claimWeek').value; const errorDiv = document.getElementById('teacherError'); const resultDiv = document.getElementById('teacherResult'); errorDiv.textContent = ''; resultDiv.classList.add('hidden'); if (!week) { errorDiv.textContent = 'Please select an academic week first from the dropdown above.'; return; } const days = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday']; let totalSessions = 0; let summaryDetails = {}; days.forEach(day => { const sessionInput = document.getElementById(`sessions_${day}`); const detailsInput = document.getElementById(`details_${day}`); const count = sessionInput ? (parseInt(sessionInput.value) || 0) : 0; const details = detailsInput ? detailsInput.value.trim() : ''; totalSessions += count; summaryDetails[day] = { count, details }; }); if (totalSessions === 0) { errorDiv.textContent = 'Please enter at least one session/lesson taught before submitting.'; return; } const ratePerSession = 500; const amount = totalSessions * ratePerSession; if (!teacherDatabase[currentLoggedInTeacher].claims) { teacherDatabase[currentLoggedInTeacher].claims = {}; } teacherDatabase[currentLoggedInTeacher].claims[week] = { week: week, totalSessions: totalSessions, amount: amount, details: summaryDetails, status: 'Pending Verification', dateSubmitted: new Date().toLocaleDateString() }; saveData(); resultDiv.classList.remove('hidden'); resultDiv.innerHTML = ` Claim Submitted Successfully!
Week: ${week}
Total Sessions: ${totalSessions}
Calculated Payout: ${amount} KSH
Status: Pending Verification `; loadDashboardData(); }
