function populateTeacherSelect() { // Populate teacher selection dropdown or list for the login portal const teacherSelect = document.getElementById('teacherSelect'); if (!teacherSelect) return; // Clear existing options except default teacherSelect.innerHTML = ''; // Example teacher list (can be expanded based on school records) const teachers = [ "Teacher Bett", // Add more teachers here as needed ]; teachers.forEach(teacher => { const option = document.createElement('option'); option.value = teacher; option.textContent = teacher; teacherSelect.appendChild(option); }); } function handleTeacherLogin(event) { if (event) event.preventDefault(); const selectedTeacher = document.getElementById('teacherSelect').value; const errorDiv = document.getElementById('teacherLoginError'); if (!selectedTeacher) { if (errorDiv) errorDiv.innerText = "Please select a teacher name to continue."; return; } // Proceed to teacher dashboard or main interface document.getElementById('teacherLoginPortal').classList.add('hidden'); document.getElementById('teacherDashboard').classList.remove('hidden'); // Initialize dashboard elements for the logged-in teacher const welcomeHeading = document.getElementById('teacherWelcomeHeader'); if (welcomeHeading) { welcomeHeading.innerText = `Welcome, ${selectedTeacher}`; } } function handleAdminLogin(event) { if (event) event.preventDefault(); const passwordInput = document.getElementById('adminPassword').value; const errorDiv = document.getElementById('adminError'); // Simple placeholder check for admin password if (passwordInput === "admin123") { document.getElementById('adminPortal').classList.add('hidden'); document.getElementById('adminDashboard').classList.remove('hidden'); } else { if (errorDiv) errorDiv.innerText = "Incorrect password. Please try again."; } } function goBackToMain(portalId) { document.getElementById(portalId).classList.add('hidden'); document.getElementById('mainMenuContainer').classList.remove('hidden'); }
